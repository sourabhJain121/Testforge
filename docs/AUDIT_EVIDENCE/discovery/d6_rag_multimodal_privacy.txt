jobs:
  testforge-ci:
    name: Autonomous Hardening & Verification
    runs-on: ubuntu-latest
    timeout-minutes: 20

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python 3.13
        uses: actions/setup-python@v5
        with:
          python-version: "3.13"
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Identify changed Python source files
        id: changed-files
        run: |
          BASE_REF="${{ github.base_ref }}"
          echo "Comparing against origin/$BASE_REF..."
          git fetch origin "$BASE_REF" --depth=1 || true
          CHANGED=$(git diff --name-only "origin/$BASE_REF...HEAD" -- '*.py' | grep -v '^tests/' || true)
          echo "Changed files: $CHANGED"
          echo "files=$CHANGED" >> "$GITHUB_OUTPUT"

      - name: Execute TestForge CLI Hardening on Changed Targets
        if: steps.changed-files.outputs.files != ''
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
          CREWAI_TESTING: "true"
          CREWAI_TRACING_ENABLED: "false"
        run: |
          mkdir -p qa_artifacts
          for file in ${{ steps.changed-files.outputs.files }}; do
            if [ -f "$file" ]; then
              echo ">>> Hardening target: $file"
              python -m testforge.cli harden "$file" --target-score 80.0 --max-iterations 3 --output-dir qa_artifacts
            fi
          done

      - name: Run Full Regression Test Suite
        env:
          CREWAI_TESTING: "true"
          CREWAI_TRACING_ENABLED: "false"
        run: |
          pytest tests/ -v --junitxml=qa_artifacts/junit_suite_results.xml

      - name: Upload OASIS SARIF Security & Mutation Report
        uses: github/codeql-action/upload-sarif@v3
        if: always()
        continue-on-error: true
        with:
          sarif_file: qa_artifacts/
          category: testforge-hardening

      - name: Archive QA Artifacts & Verification Reports
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: testforge-qa-artifacts
          path: qa_artifacts/
          retention-days: 14
